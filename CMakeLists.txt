
cmake_minimum_required(VERSION 3.16)
project(alexeev_r3_cpp VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)

add_executable(alexeev_r3_cpp src/main.cpp)
target_include_directories(alexeev_r3_cpp PRIVATE src)
target_link_libraries(alexeev_r3_cpp PRIVATE Threads::Threads)

option(ALEXEEV_SANITIZE "Build with address/undefined sanitizers" OFF)
option(ALEXEEV_ENABLE_NATIVE "Enable -march=native or /arch:AVX2 when supported" OFF)
option(ALEXEEV_ENABLE_LTO "Enable link-time optimization/interprocedural optimization" OFF)
option(ALEXEEV_ENABLE_PGO_GENERATE "Build with profile-generation instrumentation" OFF)
option(ALEXEEV_ENABLE_PGO_USE "Build using generated profile data" OFF)

if (ALEXEEV_ENABLE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if (ipo_supported)
    set_property(TARGET alexeev_r3_cpp PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  else()
    message(WARNING "IPO/LTO not supported: ${ipo_error}")
  endif()
endif()

if (MSVC)
  target_compile_options(alexeev_r3_cpp PRIVATE /O2 /W4 /permissive-)
  if (ALEXEEV_ENABLE_NATIVE)
    target_compile_options(alexeev_r3_cpp PRIVATE /arch:AVX2)
  endif()
else()
  target_compile_options(alexeev_r3_cpp PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if (ALEXEEV_ENABLE_NATIVE)
    target_compile_options(alexeev_r3_cpp PRIVATE -march=native)
  endif()
  if (ALEXEEV_ENABLE_PGO_GENERATE AND ALEXEEV_ENABLE_PGO_USE)
    message(FATAL_ERROR "Choose only one of ALEXEEV_ENABLE_PGO_GENERATE or ALEXEEV_ENABLE_PGO_USE")
  endif()
  if (ALEXEEV_ENABLE_PGO_GENERATE)
    target_compile_options(alexeev_r3_cpp PRIVATE -fprofile-generate)
    target_link_options(alexeev_r3_cpp PRIVATE -fprofile-generate)
  endif()
  if (ALEXEEV_ENABLE_PGO_USE)
    target_compile_options(alexeev_r3_cpp PRIVATE -fprofile-use -fprofile-correction)
    target_link_options(alexeev_r3_cpp PRIVATE -fprofile-use -fprofile-correction)
  endif()
endif()

if (ALEXEEV_SANITIZE AND NOT MSVC)
  target_compile_options(alexeev_r3_cpp PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(alexeev_r3_cpp PRIVATE -fsanitize=address,undefined)
endif()
